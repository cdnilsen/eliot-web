1.1 שִׁיר הַשִּׁירִים אֲשֶׁר לִשְׁלֹמֹֽה׃
1.2 יִשָּׁקֵנִי מִנְּשִׁיקוֹת פִּיהוּ כִּֽי־טוֹבִים דֹּדֶיךָ מִיָּֽיִן׃
1.3 לְרֵיחַ שְׁמָנֶיךָ טוֹבִים שֶׁמֶן תּוּרַק שְׁמֶךָ עַל־כֵּן עֲלָמוֹת אֲהֵבֽוּךָ׃
1.4 מָשְׁכֵנִי אַחֲרֶיךָ נָּרוּצָה הֱבִיאַנִי הַמֶּלֶךְ חֲדָרָיו נָגִילָה וְנִשְׂמְחָה בָּךְ נַזְכִּירָה דֹדֶיךָ מִיַּיִן מֵישָׁרִים אֲהֵבֽוּךָ׃ <sup>ס</sup>
1.5 שְׁחוֹרָה אֲנִי וְֽנָאוָה בְּנוֹת יְרוּשָׁלָ͏ִם כְּאָהֳלֵי קֵדָר כִּירִיעוֹת שְׁלֹמֹֽה׃
1.6 אַל־תִּרְאוּנִי שֶׁאֲנִי <span style="color:#0044FF">שְׁחַרְחֹרֶת</span> שֶׁשֱּׁזָפַתְנִי הַשָּׁמֶשׁ בְּנֵי אִמִּי נִֽחֲרוּ־בִי שָׂמֻנִי נֹטֵרָה אֶת־הַכְּרָמִים כַּרְמִי שֶׁלִּי לֹא נָטָֽרְתִּי׃
1.7 הַגִּידָה לִּי שֶׁאָהֲבָה נַפְשִׁי אֵיכָה תִרְעֶה אֵיכָה תַּרְבִּיץ בַּֽצָּהֳרָיִם שַׁלָּמָה אֶֽהְיֶה כְּעֹטְיָה עַל עֶדְרֵי חֲבֵרֶֽיךָ׃
1.8 אִם־לֹא תֵדְעִי לָךְ הַיָּפָה בַּנָּשִׁים צְֽאִי־לָךְ בְּעִקְבֵי הַצֹּאן וּרְעִי אֶת־<span style="color:#0044FF">גְּדִיֹּתַי</span>ִךְ עַל מִשְׁכְּנוֹת הָרֹעִֽים׃ <sup>ס</sup>
1.9 לְ<span style="color:#0044FF">סֻסָת</span>ִי בְּרִכְבֵי פַרְעֹה דִּמִּיתִיךְ רַעְיָתִֽי׃
1.10 נָאווּ לְחָיַיִךְ בַּתֹּרִים צַוָּארֵךְ בַּ<span style="color:#0044FF">חֲרוּזִֽים</span>׃
1.11 תּוֹרֵי זָהָב נַעֲשֶׂה־לָּךְ עִם <span style="color:#0044FF">נְקֻדּוֹת</span> הַכָּֽסֶף׃
1.12 עַד־שֶׁהַמֶּלֶךְ בִּמְסִבּוֹ נִרְדִּי נָתַן רֵיחֽוֹ׃
1.13 צְרוֹר הַמֹּר ׀ דּוֹדִי לִי בֵּין שָׁדַי יָלִֽין׃
1.14 אֶשְׁכֹּל הַכֹּפֶר ׀ דּוֹדִי לִי בְּכַרְמֵי עֵין גֶּֽדִי׃ <sup>ס</sup>
1.15 הִנָּךְ יָפָה רַעְיָתִי הִנָּךְ יָפָה עֵינַיִךְ יוֹנִֽים׃
1.16 הִנְּךָ יָפֶה דוֹדִי אַף נָעִים אַף־עַרְשֵׂנוּ רַעֲנָנָֽה׃
1.17 קֹרוֹת בָּתֵּינוּ אֲרָזִים <span class="ketiv">רחיטנו<span class="qere">רַהִיטֵנוּ</span></span> <span style="color:#0044FF">בְּרוֹתִֽים</span>׃
2.1 אֲנִי חֲבַצֶּלֶת הַשָּׁרוֹן שֽׁוֹשַׁנַּת הָעֲמָקִֽים׃
2.2 כְּשֽׁוֹשַׁנָּה בֵּין הַחוֹחִים כֵּן רַעְיָתִי בֵּין הַבָּנֽוֹת׃
2.3 כְּתַפּוּחַ בַּעֲצֵי הַיַּעַר כֵּן דּוֹדִי בֵּין הַבָּנִים בְּצִלּוֹ חִמַּדְתִּי וְיָשַׁבְתִּי וּפִרְיוֹ מָתוֹק לְחִכִּֽי׃
2.4 הֱבִיאַנִי אֶל־בֵּית הַיָּיִן וְדִגְלוֹ עָלַי אַהֲבָֽה׃
2.5 סַמְּכוּנִי בָּֽאֲשִׁישׁוֹת רַפְּדוּנִי בַּתַּפּוּחִים כִּי־חוֹלַת אַהֲבָה אָֽנִי׃
2.6 שְׂמֹאלוֹ תַּחַת לְרֹאשִׁי וִימִינוֹ תְּחַבְּקֵֽנִי׃
2.7 הִשְׁבַּעְתִּי אֶתְכֶם בְּנוֹת יְרוּשָׁלִַם בִּצְבָאוֹת אוֹ בְּאַיְלוֹת הַשָּׂדֶה אִם־תָּעִירוּ ׀ וְֽאִם־תְּעֽוֹרְרוּ אֶת־הָאַהֲבָה עַד שֶׁתֶּחְפָּֽץ׃ <sup>ס</sup>
2.8 קוֹל דּוֹדִי הִנֵּה־זֶה בָּא מְדַלֵּג עַל־הֶהָרִים <span style="color:#0044FF">מְקַפֵּץ</span> עַל־הַגְּבָעֽוֹת׃
2.9 דּוֹמֶה דוֹדִי לִצְבִי אוֹ לְעֹפֶר הָֽאַיָּלִים הִנֵּה־זֶה עוֹמֵד אַחַר <span style="color:#0044FF">כָּתְלֵ</span>נוּ מַשְׁגִּיחַ מִן־הַֽחֲלֹּנוֹת <span style="color:#0044FF">מֵצִיץ</span> מִן־הֽ͏ַ<span style="color:#0044FF">חֲרַכִּֽים</span>׃
2.10 עָנָה דוֹדִי וְאָמַר לִי קוּמִי לָךְ רַעְיָתִי יָפָתִי וּלְכִי־לָֽךְ׃
2.11 כִּֽי־הִנֵּה <span class="ketiv">הסתו<span class="qere">הַסְּתָיו</span></span> עָבָר הַגֶּשֶׁם חָלַף הָלַךְ לֽוֹ׃
2.12 הַנִּצָּנִים נִרְאוּ בָאָרֶץ עֵת הַ<span style="color:#0044FF">זָּמִיר</span> הִגִּיעַ וְקוֹל הַתּוֹר נִשְׁמַע בְּאַרְצֵֽנוּ׃
2.13 הַתְּאֵנָה <span style="color:#0044FF">חָֽנְטָה</span> <span style="color:#0044FF">פַגֶּי</span>הָ וְהַגְּפָנִים ׀ סְמָדַר נָתְנוּ רֵיחַ קוּמִי <span class="ketiv">לכי<span class="qere">לָךְ</span></span> רַעְיָתִי יָפָתִי וּלְכִי־לָֽךְ׃ <sup>ס</sup>
2.14 יוֹנָתִי בְּחַגְוֵי הַסֶּלַע בְּסֵתֶר הַמַּדְרֵגָה הַרְאִינִי אֶת־מַרְאַיִךְ הַשְׁמִיעִינִי אֶת־קוֹלֵךְ כִּי־קוֹלֵךְ עָרֵב וּמַרְאֵיךְ נָאוֶֽה׃ <sup>ס</sup>
2.15 אֶֽחֱזוּ־לָנוּ שֽׁוּעָלִים שֽׁוּעָלִים קְטַנִּים מְחַבְּלִים כְּרָמִים וּכְרָמֵינוּ סְמָדַֽר׃
2.16 דּוֹדִי לִי וַאֲנִי לוֹ הָרֹעֶה בַּשּׁוֹשַׁנִּֽים׃
2.17 עַד שֶׁיָּפוּחַ הַיּוֹם וְנָסוּ הַצְּלָלִים סֹב דְּמֵה־לְךָ דוֹדִי לִצְבִי אוֹ לְעֹפֶר הָאַיָּלִים עַל־הָרֵי <span style="color:#0044FF">בָֽתֶר</span>׃ <sup>ס</sup>
3.1 עַל־מִשְׁכָּבִי בַּלֵּילוֹת בִּקַּשְׁתִּי אֵת שֶׁאָהֲבָה נַפְשִׁי בִּקַּשְׁתִּיו וְלֹא מְצָאתִֽיו׃
3.2 אָקוּמָה נָּא וַאֲסוֹבְבָה בָעִיר בַּשְּׁוָקִים וּבָרְחֹבוֹת אֲבַקְשָׁה אֵת שֶׁאָהֲבָה נַפְשִׁי בִּקַּשְׁתִּיו וְלֹא מְצָאתִֽיו׃
3.3 מְצָאוּנִי הַשֹּׁמְרִים הַסֹּבְבִים בָּעִיר אֵת שֶׁאָהֲבָה נַפְשִׁי רְאִיתֶֽם׃
3.4 כִּמְעַט שֶׁעָבַרְתִּי מֵהֶם עַד שֶֽׁמָּצָאתִי אֵת שֶׁאָהֲבָה נַפְשִׁי אֲחַזְתִּיו וְלֹא אַרְפֶּנּוּ עַד־שֶׁהֲבֵיאתִיו אֶל־בֵּית אִמִּי וְאֶל־חֶדֶר הוֹרָתִֽי׃
3.5 הִשְׁבַּעְתִּי אֶתְכֶם בְּנוֹת יְרוּשָׁלִַם בִּצְבָאוֹת אוֹ בְּאַיְלוֹת הַשָּׂדֶה אִם־תָּעִירוּ ׀ וְֽאִם־תְּעֽוֹרְרוּ אֶת־הָאַהֲבָה עַד שֶׁתֶּחְפָּֽץ׃ <sup>ס</sup>
3.6 מִי זֹאת עֹלָה מִן־הַמִּדְבָּר כְּתִֽימֲרוֹת עָשָׁן מְקֻטֶּרֶת מוֹר וּלְבוֹנָה מִכֹּל <span style="color:#0044FF">אַבְקַת</span> רוֹכֵֽל׃
3.7 הִנֵּה מִטָּתוֹ שֶׁלִּשְׁלֹמֹה שִׁשִּׁים גִּבֹּרִים סָבִיב לָהּ מִגִּבֹּרֵי יִשְׂרָאֵֽל׃
3.8 כֻּלָּם אֲחֻזֵי חֶרֶב מְלֻמְּדֵי מִלְחָמָה אִישׁ חַרְבּוֹ עַל־יְרֵכוֹ מִפַּחַד בַּלֵּילּֽוֹת׃ <sup>ס</sup>
3.9 <span style="color:#0044FF">אַפִּרְיוֹן</span> עָשָׂה לוֹ הַמֶּלֶךְ שְׁלֹמֹה מֵעֲצֵי הַלְּבָנֽוֹן׃
3.10 עַמּוּדָיו עָשָׂה כֶסֶף <span style="color:#0044FF">רְפִידָת</span>וֹ זָהָב מֶרְכָּבוֹ אַרְגָּמָן תּוֹכוֹ <span style="color:#0044FF">רָצוּף</span> אַהֲבָה מִבְּנוֹת יְרוּשָׁלָֽ͏ִם׃
3.11 צְאֶינָה ׀ וּֽרְאֶינָה בְּנוֹת צִיּוֹן בַּמֶּלֶךְ שְׁלֹמֹה בָּעֲטָרָה שֶׁעִטְּרָה־לּוֹ אִמּוֹ בְּיוֹם <span style="color:#0044FF">חֲתֻנָּת</span>וֹ וּבְיוֹם שִׂמְחַת לִבּֽוֹ׃ <sup>ס</sup>
4.1 הִנָּךְ יָפָה רַעְיָתִי הִנָּךְ יָפָה עֵינַיִךְ יוֹנִים מִבַּעַד לְצַמָּתֵךְ שַׂעְרֵךְ כְּעֵדֶר הָֽעִזִּים שֶׁגָּלְשׁוּ מֵהַר גִּלְעָֽד׃
4.2 שִׁנַּיִךְ כְּעֵדֶר הַקְּצוּבוֹת שֶׁעָלוּ מִן־הָרַחְצָה שֶׁכֻּלָּם מַתְאִימוֹת וְשַׁכֻּלָה אֵין בָּהֶֽם׃
4.3 כְּחוּט הַשָּׁנִי שִׂפְתֹתַיִךְ וּ<span style="color:#0044FF">מִדְבָּרֵ</span>יךְ נָאוֶה כְּפֶלַח הָֽרִמּוֹן רַקָּתֵךְ מִבַּעַד לְצַמָּתֵֽךְ׃
4.4 כְּמִגְדַּל דָּוִיד צַוָּארֵךְ בָּנוּי לְ<span style="color:#0044FF">תַלְפִּיּוֹת</span> אֶלֶף הַמָּגֵן תָּלוּי עָלָיו כֹּל שִׁלְטֵי הַגִּבּוֹרִֽים׃
4.5 שְׁנֵי שָׁדַיִךְ כִּשְׁנֵי עֳפָרִים תְּאוֹמֵי צְבִיָּה הָרוֹעִים בַּשּׁוֹשַׁנִּֽים׃
4.6 עַד שֶׁיָּפוּחַ הַיּוֹם וְנָסוּ הַצְּלָלִים אֵלֶךְ לִי אֶל־הַר הַמּוֹר וְאֶל־גִּבְעַת הַלְּבוֹנָֽה׃
4.7 כֻּלָּךְ יָפָה רַעְיָתִי וּמוּם אֵין בָּֽךְ׃ <sup>ס</sup>
4.8 אִתִּי מִלְּבָנוֹן כַּלָּה אִתִּי מִלְּבָנוֹן תָּבוֹאִי תָּשׁוּרִי ׀ מֵרֹאשׁ אֲמָנָה מֵרֹאשׁ שְׂנִיר וְחֶרְמוֹן מִמְּעֹנוֹת אֲרָיוֹת מֵֽהַרְרֵי נְמֵרִֽים׃
4.9 לִבַּבְתִּנִי אֲחֹתִי כַלָּה לִבַּבְתִּינִי <span class="ketiv">באחד<span class="qere">בְּאַחַת</span></span> מֵעֵינַיִךְ בְּאַחַד עֲנָק מִ<span style="color:#0044FF">צַּוְּרֹנָֽי</span>ִךְ׃
4.10 מַה־יָּפוּ דֹדַיִךְ אֲחֹתִי כַלָּה מַה־טֹּבוּ דֹדַיִךְ מִיַּיִן וְרֵיחַ שְׁמָנַיִךְ מִכָּל־בְּשָׂמִֽים׃
4.11 נֹפֶת תִּטֹּפְנָה שִׂפְתוֹתַיִךְ כַּלָּה דְּבַשׁ וְחָלָב תַּחַת לְשׁוֹנֵךְ וְרֵיחַ שַׂלְמֹתַיִךְ כְּרֵיחַ לְבָנֽוֹן׃ <sup>ס</sup>
4.12 גַּן ׀ נָעוּל אֲחֹתִי כַלָּה גַּל נָעוּל מַעְיָן חָתֽוּם׃
4.13 <span style="color:#0044FF">שְׁלָחַי</span>ִךְ פַּרְדֵּס רִמּוֹנִים עִם פְּרִי מְגָדִים כְּפָרִים עִם־נְרָדִֽים׃
4.14 נֵרְדְּ ׀ וְ<span style="color:#0044FF">כַרְכֹּם</span> קָנֶה וְקִנָּמוֹן עִם כָּל־עֲצֵי לְבוֹנָה מֹר וַאֲהָלוֹת עִם כָּל־רָאשֵׁי בְשָׂמִֽים׃
4.15 מַעְיַן גַּנִּים בְּאֵר מַיִם חַיִּים וְנֹזְלִים מִן־לְבָנֽוֹן׃
4.16 עוּרִי צָפוֹן וּבוֹאִי תֵימָן הָפִיחִי גַנִּי יִזְלוּ בְשָׂמָיו יָבֹא דוֹדִי לְגַנּוֹ וְיֹאכַל פְּרִי מְגָדָֽיו׃
5.1 בָּאתִי לְגַנִּי אֲחֹתִי כַלָּה אָרִיתִי מוֹרִי עִם־בְּשָׂמִי אָכַלְתִּי יַעְרִי עִם־דִּבְשִׁי שָׁתִיתִי יֵינִי עִם־חֲלָבִי אִכְלוּ רֵעִים שְׁתוּ וְשִׁכְרוּ דּוֹדִֽים׃ <sup>ס</sup>
5.2 אֲנִי יְשֵׁנָה וְלִבִּי עֵר קוֹל ׀ דּוֹדִי דוֹפֵק פִּתְחִי־לִי אֲחֹתִי רַעְיָתִי יוֹנָתִי תַמָּתִי שֶׁרֹּאשִׁי נִמְלָא־טָל קְוֻּצּוֹתַי <span style="color:#0044FF">רְסִיסֵי</span> לָֽיְלָה׃
5.3 פָּשַׁטְתִּי אֶת־כֻּתָּנְתִּי אֵיכָכָה אֶלְבָּשֶׁנָּה רָחַצְתִּי אֶת־רַגְלַי אֵיכָכָה <span style="color:#0044FF">אֲטַנְּפֵֽ</span>ם׃
5.4 דּוֹדִי שָׁלַח יָדוֹ מִן־הַחֹר וּמֵעַי הָמוּ עָלָֽיו׃
5.5 קַמְתִּֽי אֲנִי לִפְתֹּחַ לְדוֹדִי וְיָדַי נָֽטְפוּ־מוֹר וְאֶצְבְּעֹתַי מוֹר עֹבֵר עַל כַּפּוֹת הַמַּנְעֽוּל׃
5.6 פָּתַחְתִּֽי אֲנִי לְדוֹדִי וְדוֹדִי חָמַק עָבָר נַפְשִׁי יָֽצְאָה בְדַבְּרוֹ בִּקַּשְׁתִּיהוּ וְלֹא מְצָאתִיהוּ קְרָאתִיו וְלֹא עָנָֽנִי׃
5.7 מְצָאֻנִי הַשֹּׁמְרִים הַסֹּבְבִים בָּעִיר הִכּוּנִי פְצָעוּנִי נָשְׂאוּ אֶת־רְדִידִי מֵֽעָלַי שֹׁמְרֵי הַחֹמֽוֹת׃
5.8 הִשְׁבַּעְתִּי אֶתְכֶם בְּנוֹת יְרוּשָׁלָ͏ִם אִֽם־תִּמְצְאוּ אֶת־דּוֹדִי מַה־תַּגִּידוּ לוֹ שֶׁחוֹלַת אַהֲבָה אָֽנִי׃
5.9 מַה־דּוֹדֵךְ מִדּוֹד הַיָּפָה בַּנָּשִׁים מַה־דּוֹדֵךְ מִדּוֹד שֶׁכָּכָה הִשְׁבַּעְתָּֽנוּ׃
5.10 דּוֹדִי צַח וְאָדוֹם <span style="color:#0044FF">דָּגוּל</span> מֵרְבָבָֽה׃
5.11 רֹאשׁוֹ כֶּתֶם פָּז קְוּצּוֹתָיו <span style="color:#0044FF">תַּלְתַּלִּים</span> שְׁחֹרוֹת כָּעוֹרֵֽב׃
5.12 עֵינָיו כְּיוֹנִים עַל־אֲפִיקֵי מָיִם רֹֽחֲצוֹת בֶּֽחָלָב יֹשְׁבוֹת עַל־<span style="color:#0044FF">מִלֵּֽאת</span>׃
5.13 לְחָיָו כַּעֲרוּגַת הַבֹּשֶׂם מִגְדְּלוֹת <span style="color:#0044FF">מֶרְקָחִים</span> שִׂפְתוֹתָיו שֽׁוֹשַׁנִּים נֹטְפוֹת מוֹר עֹבֵֽר׃
5.14 יָדָיו גְּלִילֵי זָהָב מְמֻלָּאִים בַּתַּרְשִׁישׁ מֵעָיו <span style="color:#0044FF">עֶשֶׁת</span> שֵׁן מְעֻלֶּפֶת סַפִּירִֽים׃
5.15 שׁוֹקָיו עַמּוּדֵי שֵׁשׁ מְיֻסָּדִים עַל־אַדְנֵי־פָז מַרְאֵהוּ כַּלְּבָנוֹן בָּחוּר כָּאֲרָזִֽים׃
5.16 חִכּוֹ מַֽמְתַקִּים וְכֻלּוֹ מַחֲ זֶה דוֹדִי וְזֶה רֵעִי בְּנוֹת יְרוּשָׁלָֽ͏ִם׃
6.1 אָנָה הָלַךְ דּוֹדֵךְ הַיָּפָה בַּנָּשִׁים אָנָה פָּנָה דוֹדֵךְ וּנְבַקְשֶׁנּוּ עִמָּֽךְ׃
6.2 דּוֹדִי יָרַד לְגַנּוֹ לַעֲרוּגוֹת הַבֹּשֶׂם לִרְעוֹת בַּגַּנִּים וְלִלְקֹט שֽׁוֹשַׁנִּֽים׃
6.3 אֲנִי לְדוֹדִי וְדוֹדִי לִי הָרֹעֶה בַּשׁוֹשַׁנִּֽים׃ <sup>ס</sup>
6.4 יָפָה אַתְּ רַעְיָתִי כְּתִרְצָה נָאוָה כִּירוּשָׁלָ͏ִם אֲיֻמָּה כַּנִּדְגָּלֽוֹת׃
6.5 הָסֵבִּי עֵינַיִךְ מִנֶּגְדִּי שֶׁהֵם הִרְהִיבֻנִי שַׂעְרֵךְ כְּעֵדֶר הָֽעִזִּים שֶׁגָּלְשׁוּ מִן־הַגִּלְעָֽד׃
6.6 שִׁנַּיִךְ כְּעֵדֶר הֽ͏ָרְחֵלִים שֶׁעָלוּ מִן־הָרַחְצָה שֶׁכֻּלָּם מַתְאִימוֹת וְשַׁכֻּלָה אֵין בָּהֶֽם׃
6.7 כְּפֶלַח הָרִמּוֹן רַקָּתֵךְ מִבַּעַד לְצַמָּתֵֽךְ׃
6.8 שִׁשִּׁים הֵמָּה מְּ וּשְׁמֹנִים פִּֽילַגְשִׁים וַעֲלָמוֹת אֵין מִסְפָּֽר׃
6.9 אַחַת הִיא יוֹנָתִי תַמָּתִי אַחַת הִיא לְאִמָּהּ בָּרָה הִיא לְיֽוֹלַדְתָּהּ רָאוּהָ בָנוֹת וַֽיְאַשְּׁרוּהָ מְלָכוֹת וּפִֽילַגְשִׁים וַֽיְהַלְלֽוּהָ׃ <sup>ס</sup>
6.10 מִי־זֹאת הַנִּשְׁקָפָה כְּמוֹ־שָׁחַר יָפָה כַלְּבָנָה בָּרָה כַּֽחַמָּה אֲיֻמָּה כַּנִּדְגָּלֽוֹת׃ <sup>ס</sup>
6.11 אֶל־גִּנַּת <span style="color:#0044FF">אֱגוֹז</span> יָרַדְתִּי לִרְאוֹת בְּאִבֵּי הַנָּחַל לִרְאוֹת הֲפָֽרְחָה הַגֶּפֶן הֵנֵצוּ הָרִמֹּנִֽים׃
6.12 לֹא יָדַעְתִּי נַפְשִׁי שָׂמַתְנִי מַרְכְּבוֹת עַמִּי־נָדִֽיב׃
6.13 שׁוּבִי שׁוּבִי הַשּׁוּלַמִּית שׁוּבִי שׁוּבִי וְנֶחֱזֶה־בָּךְ מַֽה־תֶּחֱזוּ בַּשּׁוּלַמִּית כִּמְחֹלַת הַֽמַּחֲנָֽיִם׃
7.1 מַה־יָּפוּ פְעָמַיִךְ בַּנְּעָלִים בַּת־נָדִיב <span style="color:#0044FF">חַמּוּקֵי</span> יְרֵכַיִךְ כְּמוֹ חֲלָאִים מַעֲשֵׂה יְדֵי <span style="color:#0044FF">אָמָּֽן</span>׃
7.2 שָׁרְרֵךְ אַגַּן הַ<span style="color:#0044FF">סַּהַר</span> אַל־יֶחְסַר הַ<span style="color:#0044FF">מָּזֶג</span> בִּטְנֵךְ עֲרֵמַת חִטִּים <span style="color:#0044FF">סוּגָה</span> בַּשּׁוֹשַׁנִּֽים׃
7.3 שְׁנֵי שָׁדַיִךְ כִּשְׁנֵי עֳפָרִים תָּאֳמֵי צְבִיָּֽה׃
7.4 צַוָּארֵךְ כְּמִגְדַּל הַשֵּׁן עֵינַיִךְ בְּרֵכוֹת בְּחֶשְׁבּוֹן עַל־שַׁעַר בַּת־רַבִּים אַפֵּךְ כְּמִגְדַּל הַלְּבָנוֹן צוֹפֶה פְּנֵי דַמָּֽשֶׂק׃
7.5 רֹאשֵׁךְ עָלַיִךְ כַּכַּרְמֶל וְדַלַּת רֹאשֵׁךְ כָּאַרְגָּמָן מֶלֶךְ אָסוּר בָּ<span style="color:#0044FF">רְהָטִֽים</span>׃
7.6 מַה־יָּפִית וּמַה־נָּעַמְתְּ אַהֲבָה בַּתַּֽעֲנוּגִֽים׃
7.7 זֹאת קֽוֹמָתֵךְ דָּֽמְתָה לְתָמָר וְשָׁדַיִךְ לְאַשְׁכֹּלֽוֹת׃
7.8 אָמַרְתִּי אֶעֱלֶה בְתָמָר אֹֽחֲזָה בְּ<span style="color:#0044FF">סַנְסִנָּי</span>ו וְיִֽהְיוּ־נָא שָׁדַיִךְ כְּאֶשְׁכְּלוֹת הַגֶּפֶן וְרֵיחַ אַפֵּךְ כַּתַּפּוּחִֽים׃
7.9 וְחִכֵּךְ כְּיֵין הַטּוֹב הוֹלֵךְ לְדוֹדִי לְמֵישָׁרִים <span style="color:#0044FF">דּוֹבֵב</span> שִׂפְתֵי יְשֵׁנִֽים׃
7.10 אֲנִי לְדוֹדִי וְעָלַי תְּשׁוּקָתֽוֹ׃ <sup>ס</sup>
7.11 לְכָה דוֹדִי נֵצֵא הַשָּׂדֶה נָלִינָה בַּכְּפָרִֽים׃
7.12 נַשְׁכִּימָה לַכְּרָמִים נִרְאֶה אִם פָּֽרְחָה הַגֶּפֶן פִּתַּח הַסְּמָדַר הֵנֵצוּ הָרִמּוֹנִים שָׁם אֶתֵּן אֶת־דֹּדַי לָֽךְ׃
7.13 הַֽדּוּדָאִים נָֽתְנוּ־רֵיחַ וְעַל־פְּתָחֵינוּ כָּל־מְגָדִים חֲדָשִׁים גַּם־יְשָׁנִים דּוֹדִי צָפַנְתִּי לָֽךְ׃
8.1 מִי יִתֶּנְךָ כְּאָח לִי יוֹנֵק שְׁדֵי אִמִּי אֶֽמְצָאֲךָ בַחוּץ אֶשָׁקְךָ גַּם לֹא־יָבוּזוּ לִֽי׃
8.2 אֶנְהָֽגֲךָ אֲבִֽיאֲךָ אֶל־בֵּית אִמִּי תְּלַמְּדֵנִי אַשְׁקְךָ מִיַּיִן הָ<span style="color:#0044FF">רֶקַח</span> מֵעֲסִיס רִמֹּנִֽי׃
8.3 שְׂמֹאלוֹ תַּחַת רֹאשִׁי וִֽימִינוֹ תְּחַבְּקֵֽנִי׃
8.4 הִשְׁבַּעְתִּי אֶתְכֶם בְּנוֹת יְרוּשָׁלָ͏ִם מַה־תָּעִירוּ ׀ וּֽמַה־תְּעֹֽרְרוּ אֶת־הָאַהֲבָה עַד שֶׁתֶּחְפָּֽץ׃ <sup>ס</sup>
8.5 מִי זֹאת עֹלָה מִן־הַמִּדְבָּר <span style="color:#0044FF">מִתְרַפֶּקֶת</span> עַל־דּוֹדָהּ תַּחַת הַתַּפּוּחַ עֽוֹרַרְתִּיךָ שָׁמָּה חִבְּלַתְךָ אִמֶּךָ שָׁמָּה חִבְּלָה יְלָדַֽתְךָ׃
8.6 שִׂימֵנִי כַֽחוֹתָם עַל־לִבֶּךָ כַּֽחוֹתָם עַל־זְרוֹעֶךָ כִּֽי־עַזָּה כַמָּוֶת אַהֲבָה קָשָׁה כִשְׁאוֹל קִנְאָה רְשָׁפֶיהָ רִשְׁפֵּי אֵשׁ <span style="color:#0044FF">שַׁלְהֶבֶתְיָֽה</span>׃
8.7 מַיִם רַבִּים לֹא יֽוּכְלוּ לְכַבּוֹת אֶת־הָֽאַהֲבָה וּנְהָרוֹת לֹא יִשְׁטְפוּהָ אִם־יִתֵּן אִישׁ אֶת־כָּל־הוֹן בֵּיתוֹ בָּאַהֲבָה בּוֹז יָבוּזוּ לֽוֹ׃ <sup>ס</sup>
8.8 אָחוֹת לָנוּ קְטַנָּה וְשָׁדַיִם אֵין לָהּ מַֽה־נַּעֲשֶׂה לַאֲחֹתֵנוּ בַּיּוֹם שֶׁיְּדֻבַּר־בָּֽהּ׃
8.9 אִם־חוֹמָה הִיא נִבְנֶה עָלֶיהָ טִירַת כָּסֶף וְאִם־דֶּלֶת הִיא נָצוּר עָלֶיהָ לוּחַ אָֽרֶז׃
8.10 אֲנִי חוֹמָה וְשָׁדַי כַּמִּגְדָּלוֹת אָז הָיִיתִי בְעֵינָיו כְּמוֹצְאֵת שָׁלֽוֹם׃ <sup>פ</sup>
8.11 כֶּרֶם הָיָה לִשְׁלֹמֹה בְּבַעַל הָמוֹן נָתַן אֶת־הַכֶּרֶם לַנֹּטְרִים אִישׁ יָבִא בְּפִרְיוֹ אֶלֶף כָּֽסֶף׃
8.12 כַּרְמִי שֶׁלִּי לְפָנָי הָאֶלֶף לְךָ שְׁלֹמֹה וּמָאתַיִם לְנֹטְרִים אֶת־פִּרְיֽוֹ׃
8.13 הַיוֹשֶׁבֶת בַּגַּנִּים חֲבֵרִים מַקְשִׁיבִים לְקוֹלֵךְ הַשְׁמִיעִֽינִי׃
8.14 בְּרַח ׀ דּוֹדִי וּֽדְמֵה־לְךָ לִצְבִי אוֹ לְעֹפֶר הָֽאַיָּלִים עַל הָרֵי בְשָׂמִֽים׃
